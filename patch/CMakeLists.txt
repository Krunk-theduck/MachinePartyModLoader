# Turn the .gd / .json assets into byte arrays at configure time. (MSVC rejects
# string literals over ~16 KB, so we emit {0x..,0x..} arrays instead.)
set(EMBED_CONTENT "#include \"embedded.h\"\n")
function(mpml_embed file symbol)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${file}" hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
  set(EMBED_CONTENT "${EMBED_CONTENT}extern const unsigned char ${symbol}[] = {${bytes}0x00};\nextern const size_t ${symbol}Size = sizeof(${symbol}) - 1;\n" PARENT_SCOPE)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${file}")
endfunction()
mpml_embed(bootstrap.gd    kBootstrapGd)
mpml_embed(example_mod.gd  kExampleModGd)
mpml_embed(example_mod.json kExampleModJson)
mpml_embed(example_gd.gd   kExampleLooseGd)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/embedded.cpp" CONTENT "${EMBED_CONTENT}")

add_library(mpml_patch STATIC patch.cpp pck.cpp md5.cpp gdcheck.cpp "${CMAKE_CURRENT_BINARY_DIR}/embedded.cpp")
target_include_directories(mpml_patch PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mpml_patch PUBLIC asp mpml_util mpml_settings miniz)
target_link_libraries(mpml_patch PRIVATE mpml_warnings)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/embedded.cpp" PROPERTIES GENERATED TRUE)
